        set_country_flag = wet_syl_faction_transaction

          can_join_factions = yes
          can_join_opposite_factions = yes
        }
      }
      add_to_faction = A13

        clr_country_flag = wet_syl_faction_transaction

          limit = {
            is_in_faction_with = U24
            is_subject = no

          wet_syl_clear_admission = yes
          country_event = {
            id = wet_sylvania.100
          }
        }

        }
      }
    }
    else = {
      A13 = {
        wet_syl_fail_nagash = yes
