wet_inc_join_everchosen = {
  if = {
    limit = {
      wet_inc_can_join_everchosen = yes
    }
    set_country_flag = wet_inc_faction_transaction
    wet_apply_political_permissions = yes
    random_country = {
      limit = {
        exists = yes
        is_faction_leader = yes
        has_country_flag = wet_everchosen_leader
      }
      add_to_faction = PREV
    }
    clr_country_flag = wet_inc_faction_transaction
    wet_apply_political_permissions = yes
  }
}
wet_inc_reject_other_factions = {
  if = {
    limit = {
      wet_is_daemonic_incursion = yes
      is_in_faction = yes
      OR = {
        faction_leader = {
          NOT = {
            has_country_flag = wet_everchosen_leader
          }
        }
        AND = {
          NOT = {
            has_country_flag = wet_inc_everchosen_admitted
          }
          wet_inc_bordering_everchosen = no
        }
      }
    }
    faction_leader = {
      remove_from_faction = PREV
    }
  }
  else_if = {
    limit = {
      wet_is_daemonic_incursion = yes
      is_in_faction = yes
    }
    set_country_flag = wet_inc_everchosen_admitted
  }
}
