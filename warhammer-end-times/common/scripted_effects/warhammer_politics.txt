wet_apply_political_permissions = {
 if = { limit = { wet_can_join_faction = yes } set_rule = { can_join_factions = yes can_join_opposite_factions = yes } }
 else = { set_rule = { can_join_factions = no can_join_opposite_factions = no } }
 if = { limit = { wet_can_create_faction = yes } set_rule = { can_create_factions = yes } }
 else = { set_rule = { can_create_factions = no } }
 if = { limit = { has_elections = yes }
  if = { limit = { has_government = democratic } set_politics = { ruling_party = democratic elections_allowed = no } }
  else_if = { limit = { has_government = fascism } set_politics = { ruling_party = fascism elections_allowed = no } }
  else_if = { limit = { has_government = communism } set_politics = { ruling_party = communism elections_allowed = no } }
  else_if = { limit = { has_government = neutrality } set_politics = { ruling_party = neutrality elections_allowed = no } }
  else_if = { limit = { has_government = waaagh } set_politics = { ruling_party = waaagh elections_allowed = no } }
  else_if = { limit = { has_government = horned_rat } set_politics = { ruling_party = horned_rat elections_allowed = no } }
 }
}
