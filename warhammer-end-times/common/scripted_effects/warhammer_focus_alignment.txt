wet_focus_choose_chaos = {
  if = {
    limit = {
      NOT = {
        has_government = fascism
      }
    }
    set_country_leader_ideology = wet_chaos_generic
    add_popularity = {
      ideology = fascism
      popularity = 0.60
    }
    set_politics = {
      ruling_party = fascism
      elections_allowed = no
    }
  }
  set_politics = {
    elections_allowed = no
  }
  if = {
    limit = {
      has_idea = wet_race_human
    }
    swap_ideas = {
      remove_idea = wet_race_human
      add_idea = wet_race_chaos_mortals
    }
  }
  wet_apply_political_permissions = yes
  wet_enforce_cultural_member = yes
  if = {
    limit = {
      has_global_flag = wet_population_started
    }
    wet_pop_refresh_realm = yes
  }
}
