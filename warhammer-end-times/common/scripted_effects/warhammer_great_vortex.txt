# All destructive changes are immediate, independent of event acknowledgement.
    wet_vortex_destroy_effect = {
      if = {
        limit = { wet_vortex_can_destroy = yes }
        set_global_flag = wet_story_vortex_unbound
        save_global_event_target_as = wet_vortex_destroyer
        380 = { remove_building = { type = landmark_wet_gp_great_vortex level = 1 province = 2343 } }
        set_temp_variable = { wet_vortex_factory_remainder = 0 }
        every_state = {
          limit = { wet_vortex_ulthuan_state = yes NOT = { has_state_flag = wet_vortex_flooded } }
          wet_vortex_destroy_state_factories = yes
          set_state_flag = wet_vortex_flooded
          # Thousands avoid the fixed-point overflow of large populations.
          set_temp_variable = { wet_vortex_loss_k = state_population_k }
          multiply_temp_variable = { wet_vortex_loss_k = 0.25 }
          while = {
            limit = { check_variable = { wet_vortex_loss_k > 1000 } }
            add_manpower = -1000000
            subtract_from_temp_variable = { wet_vortex_loss_k = 1000 }
          }
          multiply_temp_variable = { wet_vortex_loss_k = -1000 }
          add_manpower = var:wet_vortex_loss_k
          if = {
            limit = { NOT = { has_state_category = great_vortex } }
            add_dynamic_modifier = { modifier = wet_vortex_flooded }
          }
        }
        B99 = { add_ideas = wet_vortex_chaos_unleashed }
        every_country = {
          country_event = { id = wet_vortex.1 }
        }
      }
    }
    wet_vortex_destroy_state_factories = {
if = {
          limit = { check_variable = { industrial_complex_level > 0 } }
          set_temp_variable = { wet_vortex_factory_share = industrial_complex_level }
          multiply_temp_variable = { wet_vortex_factory_share = 0.25 }
          add_to_temp_variable = { wet_vortex_factory_remainder = wet_vortex_factory_share }
          set_temp_variable = { wet_vortex_factory_remove = wet_vortex_factory_remainder }
          round_temp_variable = wet_vortex_factory_remove
          subtract_from_temp_variable = { wet_vortex_factory_remainder = wet_vortex_factory_remove }
          if = {
            limit = { check_variable = { wet_vortex_factory_remove > 0 } }
            remove_building = { type = industrial_complex level = var:wet_vortex_factory_remove }
          }
        }
if = {
          limit = { check_variable = { arms_factory_level > 0 } }
          set_temp_variable = { wet_vortex_factory_share = arms_factory_level }
          multiply_temp_variable = { wet_vortex_factory_share = 0.25 }
          add_to_temp_variable = { wet_vortex_factory_remainder = wet_vortex_factory_share }
          set_temp_variable = { wet_vortex_factory_remove = wet_vortex_factory_remainder }
          round_temp_variable = wet_vortex_factory_remove
          subtract_from_temp_variable = { wet_vortex_factory_remainder = wet_vortex_factory_remove }
          if = {
            limit = { check_variable = { wet_vortex_factory_remove > 0 } }
            remove_building = { type = arms_factory level = var:wet_vortex_factory_remove }
          }
        }
}
