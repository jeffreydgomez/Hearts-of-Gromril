on_actions = {
 on_startup = { effect = { every_country = { wet_initialize_war_runtime_v2 = yes } } }
 # The engine may replace custom faction templates with its generic template.
 # A country flag, transferred on leadership change, identifies this faction.
 on_assume_faction_leadership = {
  effect = {
   if = {
    limit = { FROM = { has_country_flag = wet_everchosen_leader } }
    FROM = { clr_country_flag = wet_everchosen_leader }
    set_country_flag = wet_everchosen_leader
   }
  }
 }
 on_daily = {
  effect = {
   # One-time compatibility for saves made before the marker was introduced.
   if = {
    limit = { tag = B99 is_faction_leader = yes NOT = { has_global_flag = wet_everchosen_identity_initialized } }
    set_country_flag = wet_everchosen_leader
    set_global_flag = wet_everchosen_identity_initialized
    log = "WET_EVERCHOSEN_IDENTITY_INITIALIZED leader=[ROOT.GetTag]"
   }
   if = {
    limit = { has_country_flag = wet_everchosen_leader is_in_faction = no }
    clr_country_flag = wet_everchosen_leader
   }
   wet_ai_daily_war_tick = yes
  }
 }
 on_monthly = {
  effect = {
   wet_initialize_war_runtime_v2 = yes
   if = { limit = { is_ai = yes wet_archaon_rules = yes } random = { chance = 90 wet_ai_call_to_chaos = yes } }
   if = { limit = { is_ai = yes wet_name_is_greenskin = yes is_subject = no } random = { chance = 90 wet_ai_join_the_waaagh = yes } }
   if = {
    limit = {
     wet_ai_ready_for_border_war = yes
     NOT = { has_country_flag = wet_ai_war_pending }
     NOT = { has_country_flag = wet_ai_search_scheduled }
     NOT = { has_country_flag = wet_ai_war_cooldown }
    }
    if = { limit = { tag = A13 } random = { chance = 70 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { wet_name_is_greenskin = yes } random = { chance = 30 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { has_government = fascism } random = { chance = 15 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { has_government = waaagh } random = { chance = 30 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { has_government = horned_rat } random = { chance = 15 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { has_government = communism } random = { chance = 5 wet_ai_schedule_border_war_search = yes } }
    else_if = { limit = { has_government = neutrality } random = { chance = 15 wet_ai_schedule_border_war_search = yes } }
   }
   if = {
    limit = { is_ai = yes is_subject = no has_capitulated = no has_government = democratic has_war = no }
    random = { chance = 70 wet_ai_order_diplomacy = yes }
   }
  }
 }
}
