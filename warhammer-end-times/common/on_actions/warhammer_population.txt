on_actions = {
      on_startup = { effect = {
        if = { limit = { has_global_flag = wet_population_enabled NOT = { has_global_flag = wet_population_started } }
          set_global_flag = wet_population_started
          every_country = { wet_pop_update_country = yes }
          every_state = { limit = { has_state_flag = wet_inhabitants_initialized }
            wet_pop_show_race = yes wet_pop_show_faith = yes wet_pop_register = yes
          }
        }
      } }
      # ROOT = new controller; FROM = previous controller; FROM.FROM = state.
      on_state_control_changed = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_update_country = yes FROM.FROM = { wet_pop_register = yes } } } }
      on_daily_WST = { effect = { if = { limit = { has_global_flag = wet_population_started }
        clear_array = global.wet_population_remove
        for_each_scope_loop = { array = global.wet_population_active wet_pop_daily_state = yes }
        # Never remove entries while iterating the active array.
        for_each_scope_loop = { array = global.wet_population_remove remove_from_array = { global.wet_population_active = THIS } }
        clear_array = global.wet_population_remove
      } } }
      # Monthly reconciliation is one pass over controlled states in total;
      # it catches unusual scripted transfers and religious leader changes.
      on_monthly = { effect = { if = { limit = { has_global_flag = wet_population_started }
        wet_pop_refresh_realm = yes
      } } }
     on_annex = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_peaceconference_ended = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_government_change = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_ruling_party_change = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_release_as_free = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_release_as_puppet = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_liberate = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
 on_civil_war_end = { effect = { if = { limit = { has_global_flag = wet_population_started } wet_pop_refresh_realm = yes } } }
}
